void TaskLCD(void *pvParameters) {
  (void)pvParameters;
  for (;;) {
    if (xSemaphoreTake(xSerialSemaphore, (TickType_t)5) == pdTRUE) {
      xSemaphoreGive(xSerialSemaphore);
       if (S_f == 0) {
        Scrolling_screen(Y);
        Y++;
        if (Y > 5) {
          Y = 1;
        }
      }
      if(Flag==1||Flag==2&&Flag!=0){
        j++;
      }       
      else{
       j=0;
      }
      Serial.println(j);      
      if(j>50){
        S_f = 0;
        Serial.println("me");
      }
      else
      {
        S_f = 1;
      }
    }
    vTaskDelay(1000 / portTICK_PERIOD_MS);
  }
}

void curs(int c) {
  for (int i = 0; i < 4; i++) {
    lcd.setCursor(0, i);
    lcd.print(" ");
  }
  lcd.setCursor(0, c);
  lcd.print(">");
}
