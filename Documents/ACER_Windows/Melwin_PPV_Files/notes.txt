lcd2.c

#include "lcd2.h"

// Define the LCD commands and pins
#define LCD_RS_Pin GPIO_PIN_0
#define LCD_E_Pin GPIO_PIN_1
#define LCD_Data_Port GPIOA

void LCD_Enable(void) {
    HAL_GPIO_WritePin(LCD_Data_Port, LCD_E_Pin, GPIO_PIN_SET);
    HAL_Delay(1);
    HAL_GPIO_WritePin(LCD_Data_Port, LCD_E_Pin, GPIO_PIN_RESET);
    HAL_Delay(1);
}

void LCD_Send_Command(uint8_t command) {
    HAL_GPIO_WritePin(LCD_Data_Port, LCD_RS_Pin, GPIO_PIN_RESET);
    HAL_GPIO_WritePin(LCD_Data_Port, GPIO_PIN_2 | GPIO_PIN_3 | GPIO_PIN_4 | GPIO_PIN_5, (command >> 4) & 0x0F);
    LCD_Enable();
    HAL_GPIO_WritePin(LCD_Data_Port, GPIO_PIN_2 | GPIO_PIN_3 | GPIO_PIN_4 | GPIO_PIN_5, command & 0x0F);
    LCD_Enable();
}

void LCD_Init(void) {
    HAL_Delay(50);
    LCD_Send_Command(0x02);
    LCD_Send_Command(0x28);
    LCD_Send_Command(0x0C);
    LCD_Send_Command(0x06);
    LCD_Send_Command(0x01);
    HAL_Delay(5);
}

void LCD_Clear(void) {
    LCD_Send_Command(0x01);
    HAL_Delay(2);
}

void LCD_Set_Cursor(uint8_t row, uint8_t col) {
    uint8_t address;

    switch (row) {
        case 0:
            address = 0x80 + col;
            break;
        case 1:
            address = 0xC0 + col;
            break;
        default:
            address = 0x80 + col;
            break;
    }
    LCD_Send_Command(address);
}

void LCD_Write_Char(char data) {
    HAL_GPIO_WritePin(LCD_Data_Port, LCD_RS_Pin, GPIO_PIN_SET);
    HAL_GPIO_WritePin(LCD_Data_Port, GPIO_PIN_2 | GPIO_PIN_3 | GPIO_PIN_4 | GPIO_PIN_5, (data >> 4) & 0x0F);
    LCD_Enable();
    HAL_GPIO_WritePin(LCD_Data_Port, GPIO_PIN_2 | GPIO_PIN_3 | GPIO_PIN_4 | GPIO_PIN_5, data & 0x0F);
    LCD_Enable();
}

void LCD_Write_String(char *str) {
    while (*str) {
        LCD_Write_Char(*str++);
    }
}
