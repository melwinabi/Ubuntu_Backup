#include <Arduino_FreeRTOS.h>
#include <semphr.h>
#include <LiquidCrystal.h>

LiquidCrystal lcd(2, 3, 4, 5, 6, 7);
SemaphoreHandle_t xSerialSemaphore;

int menu = 12;
int up = 11;
int down = 10;
int back = 9;
int select= 8;
int S_f = 0;
int Y = 1;
int Flag = 0;
int M, U, D, B, S;
int C=0;
long int PreviousMillis = 0;
long int currentMillis;
long int Time;
void TaskButton(void *pvParameters);
void TaskLCD(void *pvParameters);
void Menu();
void Menu2();
void curs(int c);

void setup() {
  Serial.begin(9600);
  lcd.begin(20, 4);
  lcd.clear();
  pinMode(menu, INPUT_PULLUP);
  pinMode(up, INPUT_PULLUP);
  pinMode(down, INPUT_PULLUP);
  pinMode(back, INPUT_PULLUP);
  pinMode(select, INPUT_PULLUP);
  xSerialSemaphore = xSemaphoreCreateMutex();
  xTaskCreate(TaskButton, "ButtonTask", 128, NULL, 2, NULL);
  xTaskCreate(TaskLCD, "LCDTask", 128, NULL, 1, NULL);
}

void loop() {}

void TaskButton(void *pvParameters) {
  (void)pvParameters;
  for (;;) {
    
    if (xSemaphoreTake(xSerialSemaphore, (TickType_t)5) == pdTRUE) {
      M = digitalRead(menu);
      U = digitalRead(up);
      D = digitalRead(down);
      B = digitalRead(back);
      S = digitalRead(select);
      
      delay(200);
//....................................Menu 
      if (M == LOW) {
        Menu();
      }
//................................dOWN     
      if (Flag == 1 && D == LOW) 
      { 
        C++;      
        curs(C);              
      if (C == 4&&Flag == 1) 
      {
        Menu2();
        curs(C);          
      }
   }
     if(Flag==2&&D==0)
     {
        C++;
        curs(C);
        if(C>1)
        {
          C=0;
          curs(C);
        }
     }
//....................................DOWN     
     if(Flag==2&&B==0)
    {  
       C=0;
       Menu();
       
     }
     if(Flag==1&&C==3&&S==0)
     {
       //Clock();
     }
//.......................................up     
    if (Flag==1&&U==0)
    {
      C--;
      curs(C);
      if(C<=0){
        C=0;
        curs(C);
      }
     }
      if(Flag==2&&C==1&&U==0)
      { 
      C=0;
      curs(C);
      }
      if(C==0&&Flag==2&&U==0){
        C=3;
        curs(C);
        Menu();
      }
     
//..........................................uP     
//........................................Menu....................
      Serial.println(C);
      xSemaphoreGive(xSerialSemaphore);
    }
    vTaskDelay(50 / portTICK_PERIOD_MS);
  }
}


void TaskLCD(void *pvParameters) {
  (void)pvParameters;
  for (;;) {
    if (xSemaphoreTake(xSerialSemaphore, (TickType_t)5) == pdTRUE) {
      xSemaphoreGive(xSerialSemaphore);
      if (S_f == 0) {
        Scrolling_screen(Y);
        Y++;
        if (Y > 5) {
          Y = 1;
        }
      }
    }
    vTaskDelay(1000 / portTICK_PERIOD_MS);
  }
}

void Scrolling_screen(int y) {
  if (y == 1) index_1();
  if (y == 2) index_2();
  if (y == 3) index_3();
  if (y == 4) index_4();
  if (y == 5) index_5();
}

  lcd.clear();

  lcd.print("231 242 210 PN AUT");

  lcd.print("R.W.P -> Mode");
  lcd.setCursor(1, 2);
  lcd.print("1 ON | 2 OFF");
  lcd.setCursor(1, 3);
  lcd.print("NS @60 Min");
  vTaskDelay(3000 / portTICK_PERIOD_MS);
}

void curs(int c) {
  for (int i = 0; i < 4; i++) {
    lcd.setCursor(0, i);
    lcd.print(" ");
  }
  lcd.setCursor(0, c);
  lcd.print(">");
}
