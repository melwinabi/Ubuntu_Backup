#include "eeprom.h"
#include <string.h>  // Include for memset function
#include <math.h>  // Include for log function

// Function to calculate the number of bytes remaining to write within the current page
uint16_t bytestowrite(uint16_t size, uint16_t offset) {
    uint16_t space_in_page = PAGE_SIZE - offset;
    return (size < space_in_page) ? size : space_in_page;
}

HAL_StatusTypeDef EEPROM_Write(uint16_t page, uint16_t offset, uint8_t *data, uint16_t size) {
    // Calculate the bit position where the page addressing starts
    int paddrposition = log(PAGE_SIZE) / log(2);

    // Calculate the start and end pages
    uint16_t startPage = page;
    uint16_t endPage = page + ((size + offset) / PAGE_SIZE);

    // Calculate the number of pages to be written
    uint16_t numofpages = (endPage - startPage) + 1;
    uint16_t pos = 0;

    // Write the data
    for (int i = 0; i < numofpages; i++) {
        uint16_t MemAddress = (startPage << paddrposition) | offset;
        uint16_t bytesremaining = bytestowrite(size, offset);

        HAL_StatusTypeDef status = HAL_I2C_Mem_Write(EEPROM_I2C,EEPROM_ADDR , MemAddress, I2C_MEMADD_SIZE_16BIT, &data[pos], bytesremaining, 1000);

        if (status != HAL_OK) {
            return status;  // Return error status if write fails
        }

        startPage += 1;  // Increment the page for the next write
        offset = 0;  // Reset offset for new page
        size -= bytesremaining;  // Reduce the size of remaining data
        pos += bytesremaining;  // Update the position in the data buffer

        HAL_Delay(5);  // Write cycle delay (5ms)
    }

    return HAL_OK;  // Return HAL_OK if write operation is successful
}

HAL_StatusTypeDef EEPROM_Read(uint16_t page, uint16_t offset, uint8_t *data, uint16_t size) {
    int paddrposition = log(PAGE_SIZE) / log(2);

    uint16_t startPage = page;
    uint16_t endPage = page + ((size + offset) / PAGE_SIZE);

    uint16_t numofpages = (endPage - startPage) + 1;
    uint16_t pos = 0;

    for (int i = 0; i < numofpages; i++) {
        uint16_t MemAddress = (startPage << paddrposition) | offset;
        uint16_t bytesremaining = bytestowrite(size, offset);

        HAL_StatusTypeDef status = HAL_I2C_Mem_Read(EEPROM_I2C, EEPROM_ADDR, MemAddress, I2C_MEMADD_SIZE_16BIT, &data[pos], bytesremaining, 1000);

        if (status != HAL_OK) {
            return status;  // Return error status if read fails
        }

        startPage += 1;
        offset = 0;
        size -= bytesremaining;
        pos += bytesremaining;
    }

    return HAL_OK;  // Return HAL_OK if read operation is successful
}

void EEPROM_PageErase(uint16_t page) {
    int paddrposition = log(PAGE_SIZE) / log(2);
    uint16_t MemAddress = page << paddrposition;

    uint8_t data[PAGE_SIZE];
    memset(data, 0xff, PAGE_SIZE);

    HAL_I2C_Mem_Write(EEPROM_I2C, EEPROM_ADDR, MemAddress, I2C_MEMADD_SIZE_16BIT, data, PAGE_SIZE, 1000);

    HAL_Delay(5);  // Write cycle delay
}
