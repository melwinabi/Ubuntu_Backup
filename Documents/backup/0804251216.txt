float convertHexToFloat(uint8_t byte1, uint8_t byte2, uint8_t byte3, uint8_t byte4) 
{
  uint32_t combined = ((uint32_t)byte1 << 24) | ((uint32_t)byte2 << 16) | ((uint32_t)byte3 << 8) | byte4;
  union 
  {
    uint32_t i;
    float f;
  } data;
  data.i = combined;
  if (isfinite(data.f)) 
  {
    return data.f;
  } else 
  {
    return NAN;
  }
}

void LoRa_Rx() 
{
  int packetSize = LoRa.parsePacket();
  if (packetSize) 
  {
    String received = "";
    while (LoRa.available()) 
    {
      char receivedChar = (char)LoRa.read();
      received += receivedChar;
    }
    received.trim();
    //Serial.println(received);
    Lora_Handling(received);
  }
}

void PUMP_STATE(String Controll)
{
  if((Controll !=  Exe_Pst) || (TOT_Flag == true))
  {
    Exe_Pst = "";
    Exe_Pst = Controll;
    if(Controll == "11")
    {
      of_pck = 1;
      digitalWrite(RELAYON,HIGH);
      delay(5000);
      digitalWrite(RELAYON,LOW); 
      delay(100); 

      motwrk = "##*" + Master_ID + "*" + Pump_ID + "*STS*010";      
      Pmp_ACk_TGL(motwrk);
      ON_OF_CKFLG = true; 
      sts = 1;   
      Controll =="" ;
    }
    else if(Controll == "00")
    {
      if(ON_OF_CKFLG == true)
      {
        of_pck = 1;
        digitalWrite(OFRELAY,HIGH);
        delay(3500);
        digitalWrite(OFRELAY,LOW);
        delay(100);
        //Serial.println("Relay OFF"); 
        motwrk = "";
        motwrk = "##*" + Master_ID + "*" + Pump_ID + "*STS*000";      
        Pmp_ACk_TGL(motwrk);
        ON_OF_CKFLG = true;  
        sts = 0;  
        Controll =="" ;
      }        
    }
  }
}
